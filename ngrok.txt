name: Deploy to KillerKoda via Ngrok

on:
  push:
    branches: [ "main" ]

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Install dependencies
        run: npm install

      - name: Build Angular app
        run: npm run build -- --configuration production

      - name: Deploy files
        uses: appleboy/scp-action@v0.1.7
        with:
          host: ${{ secrets.NGROK_HOST }}
          port: ${{ secrets.NGROK_PORT }}
          username: ${{ secrets.EC2_USER }}
          key: ${{ secrets.EC2_KEY }}
          source: "dist/property-booking-app/*"
          target: "/var/www/angular-app/"

      - name: Restart Nginx
        uses: appleboy/ssh-action@v1.2.0
        with:
          host: ${{ secrets.NGROK_HOST }}
          port: ${{ secrets.NGROK_PORT }}
          username: ${{ secrets.EC2_USER }}
          key: ${{ secrets.EC2_KEY }}
          script: |
            sudo systemctl restart nginx


            ################################################################
NGROK_HOST = 0.tcp.ngrok.io

NGROK_PORT = 17890

EC2_USER = ubuntu (or whatever user you SSH as)

EC2_KEY = your private key
